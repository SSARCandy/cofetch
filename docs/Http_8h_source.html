200  static size_t socket_fn(CURL* e, curl_socket_t s, int epe, Http* http,
201  void* socketp) {
202  if (epe == CURL_POLL_REMOVE) {
203  epoll_ctl(http->fd_epoll_, EPOLL_CTL_DEL, s, NULL);
204  http->fd_set_.erase(s);
205  return 0;
206  }
207  //((epe & CURL_POLL_IN) ? EPOLLIN : 0) |
208  //((epe & CURL_POLL_OUT) ? EPOLLOUT : 0);
209  const static uint32_t e_table[] = {
210  0,
211  EPOLLIN,
212  EPOLLOUT,
213  EPOLLIN | EPOLLOUT,
214  };
215  epoll_event ev;
216  ev.events = e_table[epe];
217  ev.data.fd = s;
218  if (http->fd_set_.find(s) == http->fd_set_.end()) {
219  epoll_ctl(http->fd_epoll_, EPOLL_CTL_ADD, s, &ev);
220  http->fd_set_.emplace(s);
